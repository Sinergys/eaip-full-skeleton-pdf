# domain/normative_validator.py
class NormativeValidator:
    def validate_rule(self, rule: Dict[str, Any]) -> Tuple[bool, List[str]]:
        """Валидация правила, возвращает (is_valid, errors)"""
        errors = []
        
        # Валидация формулы
        if rule.get("formula"):
            if not self._validate_formula_syntax(rule["formula"]):
                errors.append("Невалидный синтаксис формулы")
        
        # Валидация единиц измерения
        if rule.get("unit"):
            if not self._validate_unit(rule["unit"]):
                errors.append(f"Неизвестная единица измерения: {rule['unit']}")
        
        # Валидация числовых значений
        if rule.get("numeric_value") is not None:
            if not self._validate_numeric_value(rule["numeric_value"], rule.get("unit")):
                errors.append("Числовое значение вне допустимого диапазона")
        
        return len(errors) == 0, errors
```

**Критерии готовности:**
- Все типы правил валидируются
- Ошибки логируются
- Невалидные правила помечаются флагом

def get_normative_value(field_name: str, sheet_name: str) -> Optional[float]:
    """Получить нормативное значение для поля"""
    rules = database.get_normative_rules_for_field(field_name, sheet_name)
    if rules:
        # Возвращаем значение с наивысшей уверенностью
        return max(rules, key=lambda r: r.get("extraction_confidence", 0)).get("numeric_value")
    return None

def validate_against_normative(actual_value: float, field_name: str, sheet_name: str) -> Dict[str, Any]:
    """Проверить соответствие фактического значения нормативу"""
    normative = get_normative_value(field_name, sheet_name)
    if normative:
        deviation = abs(actual_value - normative) / normative * 100
        return {
            "is_valid": deviation <= 10,  # 10% отклонение допустимо
            "deviation_percent": deviation,
            "normative_value": normative,
        }
    return {"is_valid": True, "message": "Норматив не найден"}
```
