⚡ КАК НАЧАТЬ РАБОТУ:

```python
from tools.unified_tasks_manager import (
    load_tasks, get_available_tasks_unified,
    lock_task_unified, unlock_task_unified,
    update_task, add_solution, add_execution
)
from tools.agent_sync_utils import (
    lock_task, unlock_task, update_task_status
)

# Загрузить задачи и выбрать P0
tasks_data = load_tasks()
available = get_available_tasks_unified(priority="P0")
if available:
    task = available[0]
    task_id = task["id"]
    
    # Заблокировать задачу (15 минут)
    if lock_task_unified(task_id, "agent_2", duration_minutes=15):
        lock_task(task_id, "agent_2", duration_minutes=15)
        update_task_status(task_id, "in_progress", "agent_2")
        update_task(task_id, {"status": "in_progress"}, "agent_2")
        
        # РАБОТАТЬ (максимум 10 минут - см. PROJECT_WORK_RULES.md!)
        # ... твой код ...
        
        # Зафиксировать выполнение
        add_execution(task_id, {
            "block_number": 1,
            "status": "in_progress",
            "completed": ["Что сделано"],
            "next_step": "Следующий шаг",
            "files_changed": ["path/to/file.py"],
            "time_spent_minutes": 10
        }, "agent_2")
        
        # Снять блокировки
        unlock_task_unified(task_id, "agent_2")
        unlock_task(task_id)
        
        # TLDV отчёт (формат см. PROJECT_WORK_RULES.md)
        print("## БЛОК 1: [Название]\n**Выполнено:**\n- [Что сделано]\n**Следующий шаг:**\n- [Что дальше]\n**Ожидание:** Ожидаю указаний пользователя.")
```
